\section{Random Search}

%-------------------------------------------------------------------------
\begin{frame}
  % If~$\func$ is differentaible, we reach the opitmizer based on the gradient information, now we don't have this information any more.
  \centering{\red{\large How can we reach the optimizer?}}
  \begin{block}{Random Search \footnote{\cite{573b8afee53b4e0eb2b0ae1cbee23c59}}}
    Having a starting point~$m_{0}\then$ generate random points around~$m_{0} \then$ calculate the function values~$\func$
  \end{block}

  This is as stated " Finding a needle in an n-dimensional haystack" \footnote{\cite{korn1966random}}.

  % Random search has limitations on the fact that they cannot move the process systematically.
  \begin{block}{Search information}
    The optimization algorithm uses a set of randomly generated points based on probability distribution.
    \end{block}
\end{frame}

\section{Evolution Strategy:~$(\parent,\offspring)$-ES}

%-------------------------------------------------------------------------
\begin{frame}
  % In case of~$\parent=1$, the evolution strategy is the special case there is only parent in the sampling step of the sample.
  % We have obtained the algorithm as below for such scenario:
  \begin{algorithm}[H]
    \caption{$(1,\offspring)$-ES Iteration: Isotropic Search}
    \DontPrintSemicolon
    \KwIn{A point $m_{0}\in\Rn$, step size~$\step>0$, population size~$\offspring$}
    % \KwOut{Updates the next center~$m_{n+1}\in\Rn$}

    \For{$k\gets~1~\KwTo~n$}{
      % \tcp{1. Sample offspring}
      \For{$i\gets~1$ \KwTo~$\offspring$}{
        Sample~$\sample_{i} \sim\normal(0,I_{\dimension})$\;
        Compute~$X_{i}\gets m_{k} + \step\sample_{i}$ 
        \tcp*{$X_{i}\sim\normal(m_{0},\step^{2}I_{\dimension}$)}
      }

      % \tcp{2. Evaluate}
      Compute fitness value~$\func(X_{1}),\dots,\func(X_{\lambda})$\;

      % \tcp{3. Select the best} 
      $i^{*}\gets\argmin_{i\in\set{1,\dots,\offspring}}\func(X_{i})$\;

      % \tcp{4. Update the center as the best candidate}
      $m_{k+1}\gets X_{i^{*}}$\;
    }
    \Return{$m_{n}$}\;
  \end{algorithm}
\end{frame}
%-------------------------------------------------------------------------


%-------------------------------------------------------------------------
\begin{frame}
  \small
  % A more stable version of the~$(\parent,\offspring)$-ES is given 
  \begin{algorithm}[H]
    \caption{$(\parent,\offspring)$-ES Iteration: Isotropic Search}
    \DontPrintSemicolon
    \KwIn{A point~$m_{0}\in\Rn$, step size~$\step>0$, population size~$\offspring$, number if selected parent~$\parent\le\offspring$, weighted factor~$\weight_{i}$ s.t.~$\sum_{i=1}^{\parent}\weight_{i}=1$.}

    \For{$k\gets~1~\KwTo~n$}{
      \For{$i\gets~1$ \KwTo~$\offspring$}{
        $\sample_{i}\sim\normal(0,I_{\dimension})$ \;
        $X_{i} = m_{k}+\step\sample_{i}$
      }

      % \tcp{2. Evaluate and rank} 
      Compute~$\func(X_{i})$ and sort offspring by fitness:
      \begin{align*}
        X_{1:\offspring}, X_{2:\offspring}, \dots, X_{\offspring:\offspring},
      \end{align*}
      where~$X_{1:\offspring}$ is the best,~$X_{2:\offspring}$ is the second-best, and so on.

      % \tcp{3. Weighted recombination} 
      Choose~$\parent$ best candidates and recombinate
      \begin{align*}
        m_{n+1} = \sum_{i=1}^{\parent} \weight_{i}X_{i:\offspring}.
      \end{align*}
    }
  \end{algorithm}
\end{frame}

\begin{frame}{Anisotropric search}

  In stead of sampling from 
  \begin{align*}
    X \sim\normal(m,\step^{2} I),
  \end{align*}
  we consider the offspring of the form 
  \begin{align*}
    X \sim \normal(m,\step^{2}\covariance)
  \end{align*}
  where~$\covariance$ is symmetric positive definite.

  What does the covariance matrix~$\covariance$ do here?
\end{frame}
%-------------------------------------------------------------------------


%-------------------------------------------------------------------------
\begin{frame}
  A picture here similar to the one that prof explained in the first day.

  The picture equals = 
  \begin{itemize}
    \item The matrix~$\covariance$ determines the shape of and orientation of the search disrtibution.
    \item The idea of covariance is relatively simple: Directions in which successful stels have repeated ly occurred should receive larger variance in the future generations.
  \end{itemize}
  \begin{block}{Search directions}
    The optimization algorithm learns  not only where to search thought the starting point~$m$ but also how to to search through~$\covariance$.
  \end{block}
\end{frame}
