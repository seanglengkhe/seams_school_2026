\section{Random Search}

%-------------------------------------------------------------------------
\begin{frame}
  % If~$\func$ is differentaible, we reach the opitmizer based on the gradient information, now we don't have this information any more.
  \centering{\red{\huge How can we reach the optimizer?}}
  
  \vspace{.4cm}
  
  \begin{block}{Random Search \footnote{\cite{573b8afee53b4e0eb2b0ae1cbee23c59}}}
    With a starting point~$m\then$ generate random points around~$m \then$ calculate the function values~$\func$.
  \end{block}

  % Random search has limitations on the fact that they cannot move the process systematically.
  \begin{block}{\red{Search information}}
    The \red{random search} uses a point~$m$ to generate \blue{random points} based on probability distribution and calculate \blue{the fitness value (function value)}.
  \end{block}

  \textbf{Drawback: No directional learning/adaptation.}

  This is as stated " Finding a needle in an n-dimensional haystack " \footnote{\cite{korn1966random}}.
\end{frame}

\section{Covariance Matrix Adaptation Evolution Strategy (CMA-ES)}

%-------------------------------------------------------------------------
\begin{frame}{}
  \begin{block}{\red{Search Information (Expected)}}
    \red{CMA-ES} = \blue{$(\parent,\offspring)$-ES} + learning/adaptation
  \end{block}
\end{frame}
%-------------------------------------------------------------------------


%-------------------------------------------------------------------------
\begin{frame}{}

  \blue{$(\parent,\offspring)$-ES} generates points
  \begin{align*}
    X \sim\normal(m,\step^{2} \red{I}).
  \end{align*}
  \red{CMA-ES} generates points 
  \begin{align*}
    X \sim \normal(m,\step^{2}\red{\covariance})
  \end{align*}
  where~\blue{$\covariance$ is symmetric positive definite}.

  \pause
  \centering{\bf\huge\red{What does the covariance matrix~$\covariance$ do here?}}
\end{frame}

\begin{frame}
  \vspace{.3cm}
  \red{\huge\bf Overall Evaluation}
  \vspace{.3cm}
    \begin{block}{When Gradient Descent is Better}
        \begin{itemize}
            \item \textbf{Isotropic \& well-conditioned problems} (e.g., Sphere).
            \item \textbf{Execution Speed:} Extremely fast due to exact analytical gradient steps.
            \item Highly scalable to large dimensions \textbf{if} the landscape is smooth and convex.
        \end{itemize}
    \end{block}
    
    \begin{block}{When CMA-ES is Better}
        \begin{itemize}
            \item \textbf{Ill-conditioned \& correlated landscapes} (e.g., Rotated Ellipsoid): Effectively adapts its covariance matrix to match the problem's topography.
            \item \textbf{Non-convex \& multimodal problems} (e.g., Rosenbrock, Rastrigin): Its stochastic nature helps escape narrow valleys and shallow local minima.
            \item Does not require gradient information (derivative-free optimization).
        \end{itemize}
    \end{block}
\end{frame}
